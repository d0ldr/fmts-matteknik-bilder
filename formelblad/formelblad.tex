\grp{Ström, spänning, effekt}
\begin{ftab}
Ström         & I = \dfrac{\Delta Q}{\Delta t}, \quad Q = I\,\Delta t\\
Enheter       & \qty{1}{A} = \qty{1}{C/s}, \quad \qty{1}{C} = \qty{1}{A.s}\\

\grp{Kretslagar}
\begin{ftab}

Serie         & R_\text{tot} = R_1 + R_2 + \dots\\

Spännings\-delning & I = \dfrac{V_1}{R_1 + R_2}\\
                   & U_2 = V_1\dfrac{R_2}{R_1 + R_2}
\end{ftab}

% ------------------------------------------------------------------
\grp{Effektanpassning}
\begin{ftab}

Lasteffekt    & P_L = U_L I_L = V_t^2\dfrac{R_L}{(R_t + R_L)^2}\\
Anpassning    & R_L = R_t \;\Rightarrow\; P_{L\text{max}} = \dfrac{V_t^2}{4R_t}
\end{ftab}

\grp{RC-krets}
\begin{ftab}
Tidskonstant  & \tau = R C\\
Differential\-ekvation & \dfrac{\mathrm{d}u}{\mathrm{d}t} + \dfrac{1}{RC}\,u = 0\\
Uppladdning   & u_C(t) = U_0\left(1 - e^{-t/RC}\right)\\
Urladdning    & u_C(t) = U_0\, e^{-t/RC}\\
Efter $\tau$  & u_C(\tau) \approx 0{,}63\,U_0
\end{ftab}

\end{ftab}
